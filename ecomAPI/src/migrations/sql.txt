CREATE DATABASE IF NOT EXISTS my_database CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE my_database;

CREATE TABLE `Users` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `email` VARCHAR(255),
  `password` VARCHAR(255),
  `firstName` VARCHAR(255),
  `lastName` VARCHAR(255),
  `address` VARCHAR(255),
  `genderId` VARCHAR(255),
  `phonenumber` VARCHAR(255),
  `image` LONGBLOB,
  `dob` VARCHAR(255),
  `isActiveEmail` BOOLEAN,
  `roleId` VARCHAR(255),
  `statusId` VARCHAR(255),
  `usertoken` VARCHAR(255),
  `is_kol` BOOLEAN,
  `kol_status` ENUM('pending', 'approved', 'rejected'),
  `kol_tier` VARCHAR(255),
  `kol_commission_rate` DECIMAL(5,2) NOT NULL DEFAULT 0.00,
  `total_sales` INT DEFAULT 0,
  `total_followers` INT,
  `facebook_link` VARCHAR(255),
  `instagram_link` VARCHAR(255),
  `tiktok_link` VARCHAR(255),
  `youtube_link` VARCHAR(255),
  `twitter_link` VARCHAR(255),
  `other_link` VARCHAR(255),
  `bank_name` VARCHAR(255),
  `bank_account_number` VARCHAR(255),
  `bank_account_name` VARCHAR(255),
  `bank_branch` VARCHAR(255),
  `createdAt` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updatedAt` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`)
);

CREATE TABLE `AffiliateClicks` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `linkId` INT NOT NULL,
  `kolId` INT NOT NULL,
  `productId` INT NOT NULL,
  `ipAddress` VARCHAR(50) DEFAULT NULL COMMENT 'Anonymized IP address',
  `userAgent` TEXT DEFAULT NULL COMMENT 'Browser/device information',
  `clickedAt` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT 'When the click occurred',
  `referrerUrl` VARCHAR(255) DEFAULT NULL COMMENT 'Referring URL',
  `geoLocation` VARCHAR(255) DEFAULT NULL,
  `converted` BOOLEAN NOT NULL DEFAULT FALSE COMMENT 'Whether this click led to a conversion',
  `conversionId` INT DEFAULT NULL COMMENT 'Reference to Order model if converted',
  PRIMARY KEY (`id`)
);

CREATE TABLE `AffiliateOrders` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `kolId` INT NOT NULL,
  `orderId` INT NOT NULL,
  `productId` INT NOT NULL,
  `linkId` INT NOT NULL,
  `revenue` DECIMAL(15,2) NOT NULL,
  `commission` DECIMAL(15,2) NOT NULL,
  `commissionRate` DECIMAL(5,2) NOT NULL,
  `status` ENUM('pending','completed','cancelled') NOT NULL,
  `confirmedAt` DATETIME DEFAULT NULL,
  `createdAt` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updatedAt` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`)
);

CREATE TABLE `drawkols` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `kolId` INT NOT NULL,
  `reviewId` INT DEFAULT NULL,
  `image_bank` LONGBLOB DEFAULT NULL,
  `amount` DECIMAL(15,2) NOT NULL,
  `status` ENUM('pending','approved','rejected','completed') NOT NULL DEFAULT 'pending',
  `bank_account_name` VARCHAR(255) NOT NULL,
  `bank_account_number` VARCHAR(255) NOT NULL,
  `bank_name` VARCHAR(255) NOT NULL,
  `bank_branch` VARCHAR(255) DEFAULT NULL,
  `withdrawAt` DATETIME DEFAULT NULL,
  `note` VARCHAR(255) DEFAULT NULL,
  `createdAt` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updatedAt` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`kolId`) REFERENCES `Users`(`id`) ON UPDATE CASCADE ON DELETE CASCADE,
  FOREIGN KEY (`reviewId`) REFERENCES `Users`(`id`) ON UPDATE CASCADE ON DELETE SET NULL
);
